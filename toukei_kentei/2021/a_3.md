## [1]
$$
\begin{align*}
M_X(s) &= E\left[e^{sX}\right]\\
&= \sum_{x=0}^{\infty}e^{sx}\cdot \frac{\lambda^x}{x!}e^{-\lambda}\\
&=\sum_{x=0}^{\infty}\frac{(\lambda e^s)^x}{x!}e^{-\lambda}\\
&=\sum_{x=0}^{\infty}\frac{(\lambda e^s)^x}{x!}e^{-\lambda e^s}\cdot e^{\lambda e^s}\cdot e^{-\lambda}\\
&=e^{\lambda(e^s-1)}\sum_{x=0}^{\infty}\frac{(\lambda e^s)^x}{x!}e^{-\lambda e^s}\\
&=e^{\lambda(e^s-1)}
\end{align*}
$$

### [2-1]
$T$ のモーメント母関数
$$
\begin{align*}
M_T(s) &= E\left[e^{s\sum_{i=1}^nx_i}\right]\\
&= E\left[e^{sx_1}e^{sx_2}\cdots e^{sx_n}\right]\\
&=E\left[e^{sx_1}\right]E\left[e^{sx_2} \right]\cdots E\left[e^{sx_n}\right]\quad &(\because X_iが独立)\\
&= e^{n\lambda(e^s-1)}\quad &(\because [1]より)\\
\end{align*}
$$
よって、$T$ は $Po(n\lambda)$ に従う

### [2-2]
$$
\begin{align*}
P(X_1, X_2, \dots, X_n|\lambda) &= \prod_{i=1}^n\frac{\lambda^{x_i}}{x_i!}e^{-\lambda}\\
&= \frac{\lambda^{\sum_{i=1}^nx_i}}{\prod_{i=1}^nx_i!}e^{-n\lambda}\\
&= e^{-n\lambda}\lambda^{t}\cdot \prod_{i=1}^nx_i!\\
&=g(t; \lambda)h(x)
\end{align*}
$$
